4. Minh chứng kỹ thuật tiêu biểu từ mã nguồn nhánh main

4.1. Cơ chế Throttle 120ms giúp kéo/phóng mặt bằng 8K mượt ở 60fps

Trong src/components/shared/zoom/ZoomableView/hooks/use-live-viewport-transform.js (dùng chung 100% ở cả 2 dự án), vị trí kéo/phóng được cập nhật trực tiếp qua MotionValue của Framer Motion, còn việc tính toán lại danh sách ô ảnh (tile culling) trong React state được giới hạn tần suất ở mức 120ms: 

const LIVE_TRANSFORM_THROTTLE_MS = 120;

useEffect(() => {
  const flush = () => {
    rafIdRef.current = null;
    lastEmitRef.current = performance.now();
    const next = readCurrent();
    setLiveTransform((prev) => {
      if (
        Math.abs(prev.x - next.x) < 0.5 &&
        Math.abs(prev.y - next.y) < 0.5 &&
        Math.abs(prev.scale - next.scale) < 0.001
      ) {
        return prev;
      }
      return next;
    });
  };
  const unsubX = motionX.on("change", scheduleEmit);
  const unsubY = motionY.on("change", scheduleEmit);
  const unsubScale = motionScale.on("change", scheduleEmit);
  return () => { unsubX(); unsubY(); unsubScale(); };
}, [enabled, motionX, motionY, motionScale]);

4.2. Bảo vệ bộ nhớ trên iOS Safari khi tải lưới ảnh phân giải cao

Trong src/hooks/useIOSMemoryTier.js và src/components/routes/building/floor-plan/deep-zoom/DeepZoomTileLayer.jsx, hệ thống kiểm tra ngưỡng cấp phát bộ nhớ của WebKit và đưa các ô ảnh vào hàng đợi giải mã tuần tự (tối đa 2 ảnh cùng lúc trên iOS) để tránh trình duyệt tự động đóng tab: 

const IOS_MAX_CONCURRENT_DECODES = 2;
let activeDecodes = 0;
const decodeQueue = [];

function pumpDecodeQueue() {
  while (activeDecodes < IOS_MAX_CONCURRENT_DECODES && decodeQueue.length > 0) {
    const job = decodeQueue.shift();
    if (job.cancelled) continue;
    activeDecodes++;
    job.start();
  }
}
